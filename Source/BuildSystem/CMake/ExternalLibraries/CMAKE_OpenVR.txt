macro(AddOpenVR)

  # Ignore call if OpenVR is disabled!
  if (XII_BUILD_OPENVR)
    set (XII_OPENVR_DIR "" CACHE PATH "Directory of the OpenVR SDK")

    include_directories ("${XII_OPENVR_DIR}/headers")

    set (XII_OPENVR_PLATFORM "")

    if (${BUILDSYSTEM_PLATFORM_WINDOWS})
      set (XII_OPENVR_PLATFORM "win")
    elseif (${BUILDSYSTEM_PLATFORM_OSX})
      set (XII_OPENVR_PLATFORM "osx")
    elseif (${BUILDSYSTEM_PLATFORM_LINUX})
      set (XII_OPENVR_PLATFORM "linux")
    endif()

    if (${BUILDSYSTEM_PLATFORM_64BIT})
      set (XII_OPENVR_PLATFORM "${XII_OPENVR_PLATFORM}64")
    else()
      set (XII_OPENVR_PLATFORM "${XII_OPENVR_PLATFORM}32")
    endif()

    link_directories ("${XII_OPENVR_DIR}/lib/${XII_OPENVR_PLATFORM}")

    SET (LINK_LIBRARIES ${LINK_LIBRARIES} openvr_api)

  endif()

endmacro()

# TODO: Currently windows only.
MACRO(CopyOpenVRDlls)
  if (XII_BUILD_OPENVR)
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${XII_OPENVR_DIR}/bin/${XII_OPENVR_PLATFORM}/openvr_api.dll"
      $<TARGET_FILE_DIR:${PROJECT_NAME}>)
  endif()
ENDMACRO()
